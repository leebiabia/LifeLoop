2. 核心数据模型（SwiftData）

DoList 模型树

SwiftData Model — DoList
@Model
class DoItem {                          // ← 基类
    var id: UUID
    var name: String
    var note: String
    var createdAt: Date
    var doType: DoType.RawValue         // "today" | "habit" | "period"
    // ↓ 仅 TodayTask 使用
    var startTime: Date?
    var deadline: Date?
    var isCompleted: Bool
    var priority: Priority.RawValue
    var tags: [Tag]
}

@Model
class TodayTask: DoItem {              // ← 今日任务（继承 DoItem）
    // 使用父类的 startTime / deadline / isCompleted
}

@Model
class HabitTask: DoItem {              // ← 打卡任务（继承 DoItem）
    var startDate: Date
    var endDate: Date?
    var reminderTime: Date?
    var cycleType: String              // "daily" | "weekly" | "monthly" | "customDates"
    var customDates: [Date]?
    var currentStreak: Int
    var totalCount: Int
    var completionRate: Double
    @Relationship(deleteRule: .cascade)
    var records: [HabitRecord]
}

@Model
class HabitRecord {                    // ← 打卡记录
    var date: Date
    var completedAt: Date
    var note: String
}

@Model
class PeriodTask: DoItem {             // ← 时期任务（继承 DoItem）
    var startDate: Date
    var endDate: Date
    var isCompleted: Bool
    // 规则：
    // 未到 startDate → 显示在 Period 页面
    // 到达 startDate → 自动出现在 Today
    // 到达 endDate → 标记 Due Today
    // 超过 endDate → 标记 Overdue
}

Goal 模型树

SwiftData Model — Goal
@Model
class GoalItem {                        // ← 基类
    var id: UUID
    var name: String
    var desc: String
    var createdAt: Date
    var goalType: GoalType.RawValue     // "progress" | "accumulation" | "lifetime"
    var color: String                   // 主题色 hex
    var icon: String                    // SF Symbol 名称
    var totalRecords: Int               // 累计记录次数
    @Relationship(deleteRule: .cascade)
    var records: [GoalRecord]
}

@Model
class ProgressGoal: GoalItem {         // ← 进度型目标
    var targetValue: Double            // 目标值（100 本书）
    var currentValue: Double           // 当前值（35 本）
    var unit: String                   // 单位（"本"）
    // 展示：双环结构
    // 内圈 = currentValue / targetValue（百分比）
    // 外圈 = totalRecords（记录次数）
}

@Model
class AccumulationGoal: GoalItem {     // ← 累计型目标
    var targetValue: Double            // 目标值（100 万）
    var currentValue: Double           // 当前累计值
    var unit: String                   // 单位（"元"）
    // 每次记录增加数值（+100, +300, +500）
    // 内圈 = currentValue / targetValue
    // 外圈 = totalRecords
}

@Model
class LifetimeGoal: GoalItem {         // ← 终身目标
    // 无目标值，无完成状态，永久存在
    // 展示：单环结构，中心显示累计次数
}

@Model
class GoalRecord {                     // ← 目标记录
    var date: Date
    var value: Double                  // 本次增加值
    var note: String
}

3. 架构层图

MVVM + Service + Repository 分层
┌─────────────────────────────────────────────────────┐
│  View Layer (SwiftUI)                                │
│  TabView → TodayView / InboxView / GoalView / ...    │
│  Components: DoubleRingView, DoCard, ...             │
│  职责：UI 渲染、用户交互、动画                        │
├─────────────────────────────────────────────────────┤
│  ViewModel Layer (@Observable)                       │
│  TodayViewModel, GoalViewModel, ...                  │
│  职责：状态管理、业务逻辑协调、数据转换                │
├─────────────────────────────────────────────────────┤
│  Service Layer                                       │
│  NotificationService, SyncService,                   │
│  StreakCalculator, ExportService                     │
│  职责：可复用的业务逻辑、跨模块能力                    │
├─────────────────────────────────────────────────────┤
│  Repository Layer                                    │
│  DoRepository, GoalRepository                        │
│  职责：数据访问抽象、查询封装                         │
├─────────────────────────────────────────────────────┤
│  Data Layer (SwiftData)                              │
│  DoItem, GoalItem, HabitRecord, GoalRecord           │
│  职责：数据持久化、CloudKit 同步（可选）               │
└─────────────────────────────────────────────────────┘

💡 设计要点：
• SwiftData 不支持继承映射到同一张表——实际实现中，DoItem 使用单表继承模式，通过 doType 字段区分子类型，所有字段在同一模型中，用 @Transient 或可选类型约束子类独有字段
• Goal 同理，GoalItem 通过 goalType 区分子类型
• Repository 层封装 @Query 和 ModelContext 操作，ViewModel 不直接依赖 SwiftData 框架
• 圆环组件是纯 SwiftUI Canvas + SpriteView，不依赖任何第三方 UI 库
